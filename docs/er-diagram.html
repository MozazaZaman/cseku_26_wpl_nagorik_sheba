    USERS ||--o{ COMPLAINTS : "submits (1:N)"
    USERS ||--o{ VOTES : "casts (1:N)"
    USERS ||--o{ NOTIFICATIONS : "receives"
    COMPLAINTS ||--o{ VOTES : "receives (M:N via votes)"
    COMPLAINTS |o--o| COMPLAINTS : "duplicate_of (merge)"
    AUTHORITIES ||--o{ STAFF : "employs"
    AUTHORITIES ||--o{ COMPLAINTS : "routed to (Agent 5)"
    STAFF |o--o{ COMPLAINTS : "handles / labels"
    COMPLAINTS ||--o{ STATUS_HISTORY : "audit trail"
    COMPLAINTS ||--o{ AGENT_LOGS : "processed by agents"
    COMPLAINTS ||--o{ NOTIFICATIONS : "triggers"

    USERS {
        int user_id PK
        string full_name
        string email UK
        string phone
        string password_hash
        string role "citizen or staff"
        datetime created_at
    }
    AUTHORITIES {
        int authority_id PK
        string name
        string type "CITY_CORP | POUROSHOVA | UNION"
        real min_lat
        real max_lat
        real min_lng
        real max_lng
        string phone
    }
    STAFF {
        int staff_id PK
        string full_name
        string email UK
        string password_hash
        string department "road|electricity|water|gas|sanitation"
        int authority_id FK
    }
    COMPLAINTS {
        int complaint_id PK
        int user_id FK
        int authority_id FK
        int assigned_staff_id FK "nullable"
        int duplicate_of_id FK "self ref, nullable"
        string title
        text description
        string category "road|electricity|water|gas|sanitation"
        string image_url "optional photo"
        real latitude
        real longitude
        string status "submitted|verified|rejected|merged|in_process|resolved"
        real priority_score
        int vote_count
        int eta_hours "deadline by staff"
        datetime created_at
        datetime resolved_at
    }
    VOTES {
        int vote_id PK
        int user_id FK
        int complaint_id FK
        datetime created_at
    }
    EMERGENCY_SERVICES {
        int service_id PK
        string name
        string type "fire|police|wasa|lged|desa|titas_gas|toilet"
        string phone
        string address
        real latitude
        real longitude
    }
    STATUS_HISTORY {
        int history_id PK
        int complaint_id FK
        string old_status
        string new_status
        text note
        string changed_by
        datetime changed_at
    }
    NOTIFICATIONS {
        int notification_id PK
        int user_id FK
        int complaint_id FK "nullable"
        string title
        text message
        bool is_read
        datetime created_at
    }
    AGENT_LOGS {
        int log_id PK
        int complaint_id FK "nullable"
        string agent_name "verifier|classifier|dedup|ranker|router"
        string decision
        text input_summary
        text output_summary
        datetime created_at
    }
🔑 No login needed → EMERGENCY_SERVICES + public complaint browsing 📝 Submit complaint → requires USER account 📍 Agent-5 routes via AUTHORITY GPS bounds ♻️ Duplicate complaint ⇒ becomes a VOTE on the original 🔒 status = in_process ⇒ voting locked + alert shown 🏷️ Staff labels: in_process / done (+ETA)
